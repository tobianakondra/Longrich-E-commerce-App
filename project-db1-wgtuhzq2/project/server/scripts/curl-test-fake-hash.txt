# Copiez-collez cette commande dans votre terminal

  -H "Content-Type: application/json" \
  -H "User-Agent: PayDunya-IPN-Test/1.0" \
  -d '{
    "data": {

      "response_text": "Transaction Found",
      "hash": "faux_hash_pour_test_de_securite_123456789abcdef",
      "invoice": {
        "token": "test_security_fake_hash",
        "total_amount": "25000",
        "description": "Test sécurité - Commande Longrich avec faux hash",
        "items": {
          "item_0": {
            "name": "Produit Longrich Test Sécurité",
            "quantity": "2",
            "unit_price": "12500",
            "total_price": "25000",
            "description": "Test de sécurité avec hash invalide"
          }
        }
      },
      "status": "completed",
      "customer": {
        "name": "Client Test Sécurité",
        "phone": "774563209",
        "email": "test-securite@longrich.com"
      },
      "receipt_url": "https://paydunya.com/test-receipt-security.pdf",
      "custom_data": {

      }
